# IronKube v0.0.0 Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a unified Kubernetes lifecycle CLI that bootstraps production-hardened clusters across k3s and kubeadm with declarative config, security profiles, and health checks — all tested.

**Architecture:** Go single binary with phase-based execution engine. Plugin interfaces for distros (k3s, kubeadm) and security profiles (minimal, cis-hardened). SSH executor for remote node operations. Declarative YAML config drives everything. No management cluster required.

**Tech Stack:** Go 1.26, cobra (CLI), golang.org/x/crypto/ssh, sigs.k8s.io/yaml, k8s.io/client-go, testify (testing)

```go
package config

import (
	"fmt"
	"os"

	"sigs.k8s.io/yaml"
)

func Load(path string) (*ClusterConfig, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("reading config: %w", err)
	}

	cfg := &ClusterConfig{}
	if err := yaml.Unmarshal(data, cfg); err != nil {
		return nil, fmt.Errorf("parsing config: %w", err)
	}

	ApplyDefaults(cfg)

	return cfg, nil
}
```

```go
package config

const (
	DefaultAPIVersion  = "ironkube.io/v1alpha1"
	DefaultKind        = "Cluster"
	DefaultUser        = "root"
	DefaultSSHPort     = 22
	DefaultProfile     = "minimal"
	DefaultCNI         = "flannel"
	DefaultPodCIDR     = "10.42.0.0/16"
	DefaultServiceCIDR = "10.43.0.0/16"
)

func ApplyDefaults(cfg *ClusterConfig) {
	if cfg.APIVersion == "" {
		cfg.APIVersion = DefaultAPIVersion
	}
	if cfg.Kind == "" {
		cfg.Kind = DefaultKind
	}
	if cfg.Spec.Security.Profile == "" {
		cfg.Spec.Security.Profile = DefaultProfile
	}
	if cfg.Spec.Networking.CNI == "" {
		cfg.Spec.Networking.CNI = DefaultCNI
	}
	if cfg.Spec.Networking.PodCIDR == "" {
		cfg.Spec.Networking.PodCIDR = DefaultPodCIDR
	}
	if cfg.Spec.Networking.SvcCIDR == "" {
		cfg.Spec.Networking.SvcCIDR = DefaultServiceCIDR
	}

	applyNodeDefaults(cfg.Spec.ControlPlane.Nodes)
	for i := range cfg.Spec.Workers {
		applyNodeDefaults(cfg.Spec.Workers[i].Nodes)
	}
}

func applyNodeDefaults(nodes []Node) {
	for i := range nodes {
		if nodes[i].User == "" {
			nodes[i].User = DefaultUser
		}
		if nodes[i].Port == 0 {
			nodes[i].Port = DefaultSSHPort
		}
	}
}
```

**Step 6: Implement validation**

Create `pkg/config/validate.go`:

```go
package config

import (
	"fmt"
	"net"
	"strings"
)

var (
	SupportedDistros  = []string{"k3s", "kubeadm"}
	SupportedProfiles = []string{"minimal", "cis-hardened"}
	SupportedCNIs     = []string{"flannel", "cilium", "calico"}
)

func Validate(cfg *ClusterConfig) error {
	var errs []string

	if cfg.Metadata.Name == "" {
		errs = append(errs, "metadata.name is required")
	}

	if !contains(SupportedDistros, cfg.Spec.Distro) {
		errs = append(errs, fmt.Sprintf("distro %q not supported (supported: %v)", cfg.Spec.Distro, SupportedDistros))
	}

	if cfg.Spec.Version == "" {
		errs = append(errs, "spec.version is required")
	}

	if len(cfg.Spec.ControlPlane.Nodes) == 0 {
		errs = append(errs, "at least one control plane node is required")
	}

	if cfg.Spec.ControlPlane.Replicas > 1 && cfg.Spec.ControlPlane.Replicas%2 == 0 {
		errs = append(errs, "HA control plane requires odd number of replicas (1, 3, 5)")
	}

	if cfg.Spec.ControlPlane.Replicas > 0 && len(cfg.Spec.ControlPlane.Nodes) != cfg.Spec.ControlPlane.Replicas {
		errs = append(errs, fmt.Sprintf("replicas (%d) must match number of control plane nodes (%d)", cfg.Spec.ControlPlane.Replicas, len(cfg.Spec.ControlPlane.Nodes)))
	}

	if !contains(SupportedProfiles, cfg.Spec.Security.Profile) {
		errs = append(errs, fmt.Sprintf("security profile %q not supported (supported: %v)", cfg.Spec.Security.Profile, SupportedProfiles))
	}

	if !contains(SupportedCNIs, cfg.Spec.Networking.CNI) {
		errs = append(errs, fmt.Sprintf("cni %q not supported (supported: %v)", cfg.Spec.Networking.CNI, SupportedCNIs))
	}

	for i, node := range cfg.Spec.ControlPlane.Nodes {
		if node.Host == "" {
			errs = append(errs, fmt.Sprintf("controlPlane.nodes[%d].host is required", i))
		}
	}

	for _, pool := range cfg.Spec.Workers {
		for i, node := range pool.Nodes {
			if node.Host == "" {
				errs = append(errs, fmt.Sprintf("workers[%s].nodes[%d].host is required", pool.Name, i))
			}
		}
	}

	if cfg.Spec.Networking.PodCIDR != "" {
		if _, _, err := net.ParseCIDR(cfg.Spec.Networking.PodCIDR); err != nil {
			errs = append(errs, fmt.Sprintf("networking.podCIDR is not a valid CIDR: %s", cfg.Spec.Networking.PodCIDR))
		}
	}

	if cfg.Spec.Networking.SvcCIDR != "" {
		if _, _, err := net.ParseCIDR(cfg.Spec.Networking.SvcCIDR); err != nil {
			errs = append(errs, fmt.Sprintf("networking.serviceCIDR is not a valid CIDR: %s", cfg.Spec.Networking.SvcCIDR))
		}
	}

	if len(errs) > 0 {
		return fmt.Errorf("config validation failed:\n  - %s", strings.Join(errs, "\n  - "))
	}
	return nil
}

func contains(slice []string, item string) bool {
	for _, s := range slice {
		if s == item {
			return true
		}
	}
	return false
}
```

```go
package cli

import (
	"fmt"

	"github.com/rohitg00/ironkube/pkg/config"
	"github.com/spf13/cobra"
)

func NewInitCmd() *cobra.Command {
	var configPath string

	cmd := &cobra.Command{
		Use:   "init",
		Short: "Bootstrap a new Kubernetes cluster",
		RunE: func(cmd *cobra.Command, args []string) error {
			cfg, err := config.Load(configPath)
			if err != nil {
				return fmt.Errorf("loading config: %w", err)
			}

			if err := config.Validate(cfg); err != nil {
				return err
			}

			fmt.Fprintf(cmd.OutOrStdout(), "Cluster %q validated successfully (distro: %s, nodes: %d)\n",
				cfg.Metadata.Name, cfg.Spec.Distro,
				len(cfg.Spec.ControlPlane.Nodes)+workerCount(cfg))
			return nil
		},
	}

	cmd.Flags().StringVarP(&configPath, "config", "c", "ironkube.yaml", "Path to cluster config file")
	return cmd
}

func workerCount(cfg *config.ClusterConfig) int {
	count := 0
	for _, pool := range cfg.Spec.Workers {
		count += len(pool.Nodes)
	}
	return count
}
```

```go
package engine

import (
	"context"
	"errors"
	"testing"

	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

type mockPhase struct {
	name    string
	runFunc func(ctx context.Context, state *State) error
	cleaned bool
}

func (m *mockPhase) Name() string { return m.name }
func (m *mockPhase) Run(ctx context.Context, state *State) error {
	if m.runFunc != nil {
		return m.runFunc(ctx, state)
	}
	return nil
}
func (m *mockPhase) Cleanup(ctx context.Context, state *State) error {
	m.cleaned = true
	return nil
}

func TestPipelineRunsAllPhases(t *testing.T) {
	var order []string
	p := NewPipeline(
		&mockPhase{name: "a", runFunc: func(_ context.Context, _ *State) error {
			order = append(order, "a")
			return nil
		}},
		&mockPhase{name: "b", runFunc: func(_ context.Context, _ *State) error {
			order = append(order, "b")
			return nil
		}},
	)

	err := p.Execute(context.Background())
	require.NoError(t, err)
	assert.Equal(t, []string{"a", "b"}, order)
}

func TestPipelineStopsOnError(t *testing.T) {
	var order []string
	p := NewPipeline(
		&mockPhase{name: "a", runFunc: func(_ context.Context, _ *State) error {

			order = append(order, "b")

	assert.Error(t, err)

func TestPipelineCleanupOnFailure(t *testing.T) {

	p := NewPipeline(
```

```go
package engine

import (
	"context"
	"fmt"
)

type Pipeline struct {
	phases    []Phase
	state     *State
	completed []string
}

func NewPipeline(phases ...Phase) *Pipeline {
	return &Pipeline{
		phases: phases,
		state:  NewState(),
	}
}

func (p *Pipeline) Execute(ctx context.Context) error {
	for _, phase := range p.phases {
		if err := ctx.Err(); err != nil {
			p.cleanup(ctx)
			return fmt.Errorf("pipeline cancelled: %w", err)
		}

		if err := phase.Run(ctx, p.state); err != nil {
			p.cleanup(ctx)
			return fmt.Errorf("phase %q failed: %w", phase.Name(), err)
		}

		p.completed = append(p.completed, phase.Name())
	}

	return nil
}

func (p *Pipeline) cleanup(ctx context.Context) {
	for i := len(p.completed) - 1; i >= 0; i-- {
		name := p.completed[i]
		for _, phase := range p.phases {
			if cp, ok := phase.(CleanupPhase); ok && phase.Name() == name {
				_ = cp.Cleanup(ctx, p.state)
			}
		}
	}
}

func (p *Pipeline) CompletedPhases() []string {
	return p.completed
}

func (p *Pipeline) State() *State {
	return p.state
}
```

```go
package ssh

import (
	"fmt"
	"io"
	"net"
	"os"
	"time"

	gossh "golang.org/x/crypto/ssh"
)

type Client struct {
	conn *gossh.Client
	host string
}

type Config struct {
	Host       string
	Port       int
	User       string
	KeyPath    string
	Timeout    time.Duration
}

func NewClient(cfg Config) (*Client, error) {
	if cfg.Timeout == 0 {
		cfg.Timeout = 30 * time.Second
	}

	auth, err := authMethod(cfg.KeyPath)
	if err != nil {
		return nil, fmt.Errorf("auth setup: %w", err)
	}

	sshCfg := &gossh.ClientConfig{
		User:            cfg.User,
		Auth:            []gossh.AuthMethod{auth},
		HostKeyCallback: gossh.InsecureIgnoreHostKey(),
		Timeout:         cfg.Timeout,
	}

	addr := net.JoinHostPort(cfg.Host, fmt.Sprintf("%d", cfg.Port))
	conn, err := gossh.Dial("tcp", addr, sshCfg)
	if err != nil {
		return nil, fmt.Errorf("connecting to %s: %w", addr, err)
	}

	return &Client{conn: conn, host: cfg.Host}, nil
}

func (c *Client) Run(cmd string) (string, error) {
	session, err := c.conn.NewSession()
	if err != nil {
		return "", fmt.Errorf("creating session: %w", err)
	}
	defer session.Close()

	output, err := session.CombinedOutput(cmd)
	if err != nil {
		return string(output), fmt.Errorf("running %q on %s: %w\nOutput: %s", cmd, c.host, err, output)
	}
	return string(output), nil
}

func (c *Client) RunStream(cmd string, stdout, stderr io.Writer) error {
	session, err := c.conn.NewSession()
	if err != nil {
		return fmt.Errorf("creating session: %w", err)
	}
	defer session.Close()

	session.Stdout = stdout
	session.Stderr = stderr
	return session.Run(cmd)
}

func (c *Client) Close() error {
	return c.conn.Close()
}

func (c *Client) Host() string {
	return c.host
}

func authMethod(keyPath string) (gossh.AuthMethod, error) {
	if keyPath == "" {
		keyPath = os.ExpandEnv("$HOME/.ssh/id_rsa")
	}

	key, err := os.ReadFile(keyPath)
	if err != nil {
		return nil, fmt.Errorf("reading key %s: %w", keyPath, err)
	}

	signer, err := gossh.ParsePrivateKey(key)
	if err != nil {
		return nil, fmt.Errorf("parsing key: %w", err)
	}

	return gossh.PublicKeys(signer), nil
}
```

```go
package ssh

import (
	"context"
	"fmt"
	"sync"
)

type Result struct {
	Host   string
	Output string
	Err    error
}

type Executor struct {
	clients map[string]*Client
	mu      sync.RWMutex
}

func NewExecutor() *Executor {
	return &Executor{
		clients: make(map[string]*Client),
	}
}

func (e *Executor) Connect(cfg Config) error {
	client, err := NewClient(cfg)
	if err != nil {
		return err
	}

	e.mu.Lock()
	defer e.mu.Unlock()
	e.clients[cfg.Host] = client
	return nil
}

func (e *Executor) RunOnAll(ctx context.Context, cmd string) []Result {
	e.mu.RLock()
	hosts := make([]string, 0, len(e.clients))
	for host := range e.clients {
		hosts = append(hosts, host)
	}
	e.mu.RUnlock()

	return e.RunOnHosts(ctx, hosts, cmd)
}

func (e *Executor) RunOnHosts(ctx context.Context, hosts []string, cmd string) []Result {
	results := make([]Result, len(hosts))
	var wg sync.WaitGroup

	for i, host := range hosts {
		wg.Add(1)
		go func(idx int, h string) {
			defer wg.Done()

			if err := ctx.Err(); err != nil {
				results[idx] = Result{Host: h, Err: err}
				return
			}

			e.mu.RLock()
			client, ok := e.clients[h]
			e.mu.RUnlock()

			if !ok {
				results[idx] = Result{Host: h, Err: fmt.Errorf("no connection to %s", h)}
				return
			}

			output, err := client.Run(cmd)
			results[idx] = Result{Host: h, Output: output, Err: err}
		}(i, host)
	}

	wg.Wait()
	return results
}

func (e *Executor) RunOnHost(host, cmd string) (string, error) {
	e.mu.RLock()
	client, ok := e.clients[host]
	e.mu.RUnlock()

	if !ok {
		return "", fmt.Errorf("no connection to %s", host)
	}
	return client.Run(cmd)
}

func (e *Executor) Close() {
	e.mu.Lock()
	defer e.mu.Unlock()
	for _, client := range e.clients {
		client.Close()
	}
	e.clients = make(map[string]*Client)
}
```

```go
import (
	"fmt"
	"strings"

	"github.com/rohitg00/ironkube/pkg/config"
)

type Plugin struct{}

func New() *Plugin {
	return &Plugin{}
}

func (p *Plugin) Name() string { return "k3s" }

func (p *Plugin) ValidateVersion(version string) error {
	if version == "" {
		return fmt.Errorf("version is required")
	}
	if !strings.HasPrefix(version, "v") {
		return fmt.Errorf("version must start with 'v': %s", version)
	}
	if !strings.Contains(version, "+k3s") {
		return fmt.Errorf("k3s version must contain '+k3s': %s", version)
	}
	return nil
}

func (p *Plugin) ServerInstallScript(node config.Node, cfg *config.ClusterConfig, token string, isInit bool) string {
	var sb strings.Builder

	sb.WriteString(fmt.Sprintf("export INSTALL_K3S_VERSION=%s\n", cfg.Spec.Version))

	if !isInit && token != "" {
		sb.WriteString(fmt.Sprintf("export K3S_URL=%s\n", token))
	}

	sb.WriteString("curl -sfL https://get.k3s.io | sh -s - server")

	if cfg.Spec.ControlPlane.Replicas > 1 && isInit {
		sb.WriteString(" --cluster-init")
	}

	if !isInit && token != "" {
		sb.WriteString(fmt.Sprintf(" --token ${K3S_TOKEN}"))
	}

	sb.WriteString(fmt.Sprintf(" --cluster-cidr %s", cfg.Spec.Networking.PodCIDR))
	sb.WriteString(fmt.Sprintf(" --service-cidr %s", cfg.Spec.Networking.SvcCIDR))
	sb.WriteString(" --write-kubeconfig-mode 644")

	return sb.String()
}

func (p *Plugin) AgentInstallScript(node config.Node, cfg *config.ClusterConfig, serverURL string, token string) string {
	var sb strings.Builder

	sb.WriteString(fmt.Sprintf("export INSTALL_K3S_VERSION=%s\n", cfg.Spec.Version))
	sb.WriteString(fmt.Sprintf("export K3S_URL=%s\n", serverURL))
	sb.WriteString(fmt.Sprintf("export K3S_TOKEN=%s\n", token))
	sb.WriteString("curl -sfL https://get.k3s.io | sh -")

	return sb.String()
}
```

```go
package kubeconfig

import (
	"fmt"
	"os"
	"strings"

	"sigs.k8s.io/yaml"
)

type Config struct {
	APIVersion     string    `yaml:"apiVersion" json:"apiVersion"`
	Kind           string    `yaml:"kind" json:"kind"`
	Clusters       []Cluster `yaml:"clusters" json:"clusters"`
	Contexts       []Context `yaml:"contexts" json:"contexts"`
	CurrentContext string    `yaml:"current-context" json:"current-context"`
	Users          []User    `yaml:"users" json:"users"`
}

type Cluster struct {
	Name    string      `yaml:"name" json:"name"`
	Cluster ClusterData `yaml:"cluster" json:"cluster"`
}

type ClusterData struct {
	Server                   string `yaml:"server" json:"server"`
	CertificateAuthorityData string `yaml:"certificate-authority-data,omitempty" json:"certificate-authority-data,omitempty"`
}

type Context struct {
	Name    string      `yaml:"name" json:"name"`
	Context ContextData `yaml:"context" json:"context"`
}

type ContextData struct {
	Cluster string `yaml:"cluster" json:"cluster"`
	User    string `yaml:"user" json:"user"`
}

type User struct {
	Name string   `yaml:"name" json:"name"`
	User UserData `yaml:"user" json:"user"`
}

type UserData struct {
	ClientCertificateData string `yaml:"client-certificate-data,omitempty" json:"client-certificate-data,omitempty"`
	ClientKeyData         string `yaml:"client-key-data,omitempty" json:"client-key-data,omitempty"`
	Token                 string `yaml:"token,omitempty" json:"token,omitempty"`
}

func RewriteServer(data []byte, host string, clusterName string) ([]byte, error) {
	cfg := &Config{}
	if err := yaml.Unmarshal(data, cfg); err != nil {
		return nil, fmt.Errorf("parsing kubeconfig: %w", err)
	}

	for i := range cfg.Clusters {
		server := cfg.Clusters[i].Cluster.Server
		if strings.Contains(server, "127.0.0.1") || strings.Contains(server, "localhost") {
			port := "6443"
			if parts := strings.Split(server, ":"); len(parts) == 3 {
				port = parts[2]
			}
			cfg.Clusters[i].Cluster.Server = fmt.Sprintf("https://%s:%s", host, port)
		}
		cfg.Clusters[i].Name = clusterName
	}

	for i := range cfg.Contexts {
		cfg.Contexts[i].Name = clusterName
		cfg.Contexts[i].Context.Cluster = clusterName
		cfg.Contexts[i].Context.User = clusterName
	}

	for i := range cfg.Users {
		cfg.Users[i].Name = clusterName
	}

	cfg.CurrentContext = clusterName

	return yaml.Marshal(cfg)
}

func SaveToFile(data []byte, path string) error {
	return os.WriteFile(path, data, 0600)
}

func Merge(existingPath string, newData []byte) error {
	existingData, err := os.ReadFile(existingPath)
	if err != nil {
		return SaveToFile(newData, existingPath)
	}

	existing := &Config{}
	if err := yaml.Unmarshal(existingData, existing); err != nil {
		return fmt.Errorf("parsing existing kubeconfig: %w", err)
	}

	newCfg := &Config{}
	if err := yaml.Unmarshal(newData, newCfg); err != nil {
		return fmt.Errorf("parsing new kubeconfig: %w", err)
	}

	existing.Clusters = appendUnique(existing.Clusters, newCfg.Clusters)
	existing.Contexts = appendUniqueCtx(existing.Contexts, newCfg.Contexts)
	existing.Users = appendUniqueUser(existing.Users, newCfg.Users)
	existing.CurrentContext = newCfg.CurrentContext

	merged, err := yaml.Marshal(existing)
	if err != nil {
		return fmt.Errorf("marshaling merged kubeconfig: %w", err)
	}

	return SaveToFile(merged, existingPath)
}

func appendUnique(existing, new []Cluster) []Cluster {
	names := make(map[string]bool)
	for _, c := range existing {
		names[c.Name] = true
	}
	for _, c := range new {
		if !names[c.Name] {
			existing = append(existing, c)
		}
	}
	return existing
}

func appendUniqueCtx(existing, new []Context) []Context {
	names := make(map[string]bool)
	for _, c := range existing {
		names[c.Name] = true
	}
	for _, c := range new {
		if !names[c.Name] {
			existing = append(existing, c)
		}
	}
	return existing
}

func appendUniqueUser(existing, new []User) []User {
	names := make(map[string]bool)
	for _, u := range existing {
		names[u.Name] = true
	}
	for _, u := range new {
		if !names[u.Name] {
			existing = append(existing, u)
		}
	}
	return existing
}
```

```go
package health

type Status string

const (
	StatusHealthy   Status = "healthy"
	StatusUnhealthy Status = "unhealthy"
	StatusUnknown   Status = "unknown"
)

type CheckResult struct {
	Name    string `json:"name"`
	Status  Status `json:"status"`
	Message string `json:"message"`
}

type ClusterHealth struct {
	ClusterName string        `json:"clusterName"`
	Results     []CheckResult `json:"results"`
}

type HealthSummary struct {
	Healthy   int `json:"healthy"`
	Unhealthy int `json:"unhealthy"`
	Unknown   int `json:"unknown"`
	Total     int `json:"total"`
}

func (ch *ClusterHealth) IsHealthy() bool {
	for _, r := range ch.Results {
		if r.Status != StatusHealthy {
			return false
		}
	}
	return true
}

func (ch *ClusterHealth) Summary() HealthSummary {
	s := HealthSummary{Total: len(ch.Results)}
	for _, r := range ch.Results {
		switch r.Status {
		case StatusHealthy:
			s.Healthy++
		case StatusUnhealthy:
			s.Unhealthy++
		default:
			s.Unknown++
		}
	}
	return s
}
```

```go
			)

			if dryRun {
				fmt.Fprintf(cmd.OutOrStdout(), "Dry run: cluster %q validated (distro: %s, control-plane: %d, workers: %d)\n",
					cfg.Metadata.Name, cfg.Spec.Distro,
					len(cfg.Spec.ControlPlane.Nodes), workerCount(cfg))
				return nil
			}

			if err := pipeline.Execute(context.Background()); err != nil {
				return err
			}

			fmt.Fprintf(cmd.OutOrStdout(), "Cluster %q initialized successfully\n", cfg.Metadata.Name)
			return nil
		},
	}

	cmd.Flags().StringVarP(&configPath, "config", "c", "ironkube.yaml", "Path to cluster config file")
	cmd.Flags().BoolVar(&dryRun, "dry-run", false, "Validate config without bootstrapping")
	return cmd
}

func workerCount(cfg *config.ClusterConfig) int {
	count := 0
	for _, pool := range cfg.Spec.Workers {
		count += len(pool.Nodes)
	}
	return count
}
```
